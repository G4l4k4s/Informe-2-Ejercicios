import java.util.Scaner;
public class SumaPares{String[] args){
    Scanner scanner= new Scanner(System.in);
    
    int n=scanner.nextInt();
    int acumulador=0;
    
    for (int i=0; i<n; i++){
        int numero = scanner.nextInt();
        if(numero%2==0){
            acumulador += numero;
        }
    }
    
    System.out.println(acumulador);
    scanner.close();
    
    
}
}
